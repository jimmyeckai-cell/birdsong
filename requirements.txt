# Bird-identification pipeline dependencies.
# Reproduce with:  python3 -m venv venv && ./venv/bin/pip install -r requirements.txt
#
# birdnetlib bundles the BirdNET tflite model (no API key / model download).
# It needs a TFLite runtime, which comes from EITHER:
#   - tensorflow (used here; has arm64 macOS + Linux wheels), OR
#   - tflite-runtime (smaller; good on aarch64 Linux where a tflite-runtime
#     wheel exists). See the numpy note below if you use tflite-runtime.
#
# NUMPY NOTE: tflite-runtime 2.14.0 was built against numpy 1.x and crashes
# under numpy 2.x ("AttributeError: _ARRAY_API not found"). If you switch to
# tflite-runtime, pin `numpy<2` (e.g. numpy==1.26.4). Full tensorflow (below)
# works with numpy 2.x, which is why we do not pin it on this install.

birdnetlib==0.18.1
librosa==0.11.0
tensorflow==2.20.0

# Watercolor bird portraits for the landing page (fetch_watercolors.py):
# Pillow ships with tensorflow's deps; opencv provides the painterly filters.
opencv-python-headless==5.0.0.93

# Optional: auto-generate watercolor art via the OpenAI Images API
# (generate_art_openai.py). Needs an OPENAI_API_KEY (platform.openai.com,
# billed per image) — this is NOT the same as a ChatGPT Pro subscription.
openai>=1.0
